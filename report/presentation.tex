\section{The Pitch}
% ====================================================================

\begin{frame}{The Problem}
\begin{columns}
\column{0.6\textwidth}
\textbf{DJI drone owners struggle with:}
\begin{itemize}
  \item Dense 100+ page PDF manuals (per model)
  \item Cryptic LED blink patterns and error codes
  \item Pricing scattered across 4+ retailers
  \item No single tool that answers all questions
\end{itemize}

\vspace{1em}
\textbf{Current solutions:}
\begin{itemize}
  \item Google search -- unreliable, outdated
  \item Reddit/forums -- anecdotal, unverified
  \item DJI support -- slow response times
\end{itemize}

\column{0.4\textwidth}
\centering
\vspace{1em}

\small
\textit{``What does 3 red blinks mean on my Mini 4 Pro?''}

\vspace{0.5em}
\textit{``Is the Air 3 cheaper on Amazon or B\&H?''}

\vspace{0.5em}
\textit{``How do I calibrate the gimbal?''}
\end{columns}
\end{frame}

\begin{frame}{The Solution}
\begin{center}
\Large\textbf{An AI-powered DJI drone assistant}\\[0.5em]
\normalsize that combines manual search, diagnostics, and live pricing\\in one conversational interface.
\end{center}

\vspace{1em}

\begin{columns}
\column{0.33\textwidth}
\centering
\textbf{RAG Agent}\\[0.3em]
\small Searches official manuals\\
Returns specs + diagrams\\
Cites source and page

\column{0.33\textwidth}
\centering
\textbf{Diagnostic Agent}\\[0.3em]
\small Resolves error codes\\
LED light patterns\\
Troubleshooting steps

\column{0.33\textwidth}
\centering
\textbf{Pricing Agent}\\[0.3em]
\small Live vendor search\\
Cross-retailer comparison\\
Stock and delivery info
\end{columns}
\end{frame}

\begin{frame}{Who Is It For?}
\begin{itemize}
  \item \textbf{Hobbyist pilots} -- quick answers about features, specs, how-to
  \item \textbf{New owners} -- troubleshooting first-flight issues
  \item \textbf{Buyers} -- comparing models and finding best prices
  \item \textbf{DJI support teams} -- internal knowledge base tool
  \item \textbf{Repair shops} -- error code resolution and parts lookup
\end{itemize}

\vspace{1em}
\textbf{Supported models:} Mini 4 Pro, Air 3, Mavic 3 Classic, Neo, and any new model ingested by admin.
\end{frame}

\begin{frame}{Key Features}
\begin{columns}

\begin{frame}[fragile]{System Architecture}
\begin{verbatim}
  [Frontend React:3000]
          |
  [agent-system-a:8000] (LangGraph + FastAPI)
      |          |           |
  [mcp-server   [agent-     [MongoDB Atlas]
   :8002]        system-b
      |          :8001]
  [Qdrant       [SerpAPI
   :6333]       Google]
      |
  [Redis:6380]
\end{verbatim}

\small
\textbf{6 containers}, all communicating over HTTP.\\
Agent B is \textbf{never imported} -- only called via network.
\end{frame}

\begin{frame}[fragile]{Pipeline Flow}
\begin{verbatim}
  User Query
      |
  [Input Guard] -- LLM detects prompt injection
      |
  [BERT Classifier] -- 50ms intent classification
      |
  [Supervisor] -- Routes (BERT confident? Skip LLM call)
      |
  [Specialist Agent] -- RAG / Diagnostic / Pricing
      |
  [Summarizer] -- Grounded Markdown response
      |
  [Output Guard] -- Sanitization
      |
  Response (with S3 images, source citations)
\end{verbatim}
\end{frame}

\begin{frame}{MCP Server -- Tools}
\textbf{3 MCP tools exposed via FastMCP:}

\vspace{0.5em}

\begin{enumerate}
  \item \textbf{query\_dji\_manual\_vector\_db}\\
  \small Hybrid search (dense + BM25) $\rightarrow$ parent dedup $\rightarrow$ rerank $\rightarrow$ top-4 parents
  
  \vspace{0.3em}
  \item \textbf{lookup\_dji\_error\_code\_db}\\
  \small Exact-match error code lookup with resolution steps and severity
  
  \vspace{0.3em}
  \item \textbf{ingest\_and\_index\_pdf} (Admin)\\
  \small PDF $\rightarrow$ text extract $\rightarrow$ GPT-4o image captions $\rightarrow$ S3 upload $\rightarrow$ parent-child chunking $\rightarrow$ Qdrant
\end{enumerate}
\end{frame}

\begin{frame}[fragile]{BERT Classifier as Routing Optimizer}
\begin{columns}
\column{0.5\textwidth}
\textbf{Purpose:} Fast pre-filter before LLM supervisor

\vspace{0.5em}
\begin{itemize}
  \item 300 training examples (3 classes)
  \item DistilBERT fine-tuned, 10 epochs
  \item 100\% test accuracy
  \item $\sim$50ms inference (local, free)
\end{itemize}

\vspace{0.5em}
\textbf{Saves:} $\sim$2s + \$0.001 per query\\by skipping LLM supervisor call

\column{0.5\textwidth}
\textbf{Routing Logic:}
\begin{verbatim}
if BERT confidence >= 0.85:
    route directly (fast path)
else:
    LLM supervisor decides
    (slow path, edge cases)
\end{verbatim}

\vspace{0.5em}
\small
\textbf{Improvement:}\\
100 samples $\rightarrow$ 95\% accuracy\\
300 samples $\rightarrow$ 100\% accuracy
\end{columns}
\end{frame}

\begin{frame}[fragile]{Agent System B -- Autonomous Pricing Agent}
\textbf{Pattern:} ReAct agent with tool-calling loop (max 8 iterations)

\vspace{0.5em}
\textbf{Tools:}
\begin{itemize}
  \item \texttt{get\_reference\_pricing} -- cached baseline data
  \item \texttt{search\_duckduckgo} -- Google via SerpAPI
  \item \texttt{compare\_prices} -- cross-vendor analysis
  \item \texttt{check\_stock\_and\_delivery} -- availability check
\end{itemize}

\vspace{0.5em}
\textbf{Flow:}
\begin{verbatim}
LLM decides -> calls tool -> gets result -> decides next
-> calls another tool -> ... -> generates final JSON
\end{verbatim}

\vspace{0.3em}
\small The LLM \textbf{autonomously} chooses which tools, in what order, and when to stop.
\end{frame}

\begin{frame}{RAG Optimization -- 9 Experiments}
\begin{table}
\centering
\tiny
\begin{tabular}{llcccc}
\toprule
\textbf{\#} & \textbf{Change} & \textbf{Faith} & \textbf{Relev} & \textbf{Prec} & \textbf{Recall} \\
\midrule
1 & Baseline (eval bug) & 0.87 & 0.87 & 0.63 & 0.90 \\
2 & Fix eval top\_k & 0.89 & 0.81 & 0.79 & 0.83 \\
3 & top\_k=4 & 0.925 & 0.865 & 0.814 & 0.855 \\
4 & Temperature 0.1 & 0.913 & 0.86 & 0.779 & 0.83 \\
5 & Semantic chunking (headers) & 0.86 & 0.71 & 0.72 & 0.73 \\

7 & BGE reranker & 0.922 & 0.857 & \textbf{0.924} & 0.875 \\
8 & Hybrid (BM25+dense) & 0.854 & 0.867 & 0.940 & 0.908 \\
\textbf{9} & \textbf{Parent-child + hybrid} & \textbf{0.958} & \textbf{0.922} & \textbf{0.936} & \textbf{0.933} \\
\bottomrule
\end{tabular}
\end{table}

\vspace{0.5em}
\textbf{Winner:} Parent-child chunking (1500/300) + hybrid search + BGE reranker
\end{frame}

\begin{frame}{What Worked vs What Failed}
\begin{columns}
\column{0.5\textwidth}
\textbf{What Worked:}
\begin{itemize}
  \item Upgraded reranker (+11\% precision)
  \item Parent-child chunking (+all metrics)
  \item Hybrid search (+recall)
  \item Reducing top\_k 5$\rightarrow$4 (+faithfulness)
\end{itemize}

\column{0.5\textwidth}
\textbf{What Failed:}
\begin{itemize}
  \item Semantic chunking (headers): -10\%
  \item Semantic chunking (embeddings): -30\%
  \item Temperature tuning: negligible
\end{itemize}

\vspace{0.5em}
\small\textit{Lesson: DJI manuals are spec tables and bullet lists -- not prose. Fixed-length + overlap is more robust than ``smart'' splitting.}
\end{columns}
\end{frame}

\section{Failure Cases and Production Thinking}
% ====================================================================

\begin{frame}{Documented Failure Cases}
\small
\begin{enumerate}
  \item \textbf{BM25 stale after ingestion} (Design) -- Fixed: auto-rebuild after ingest/delete
  \item \textbf{Metadata filter mismatch} (Prompt) -- Fixed: strict model names in planner prompt
  \item \textbf{DuckDuckGo rate-limiting} (External) -- Fixed: replaced with SerpAPI
  \item \textbf{Cache disabled for filtered queries} (Design) -- Fixed: model-aware cache keys
  \item \textbf{Multi-model comparison failure} (Design) -- Fixed: parallel filtered searches
  \item \textbf{Pricing history pollution} (Design) -- Fixed: query-first extraction
\end{enumerate}

\vspace{0.5em}
\textit{Each failure was diagnosed, root-caused (model/prompt/design/external), and resolved with a targeted fix.}
\end{frame}

\begin{frame}{Production Features}
\begin{columns}
\column{0.5\textwidth}
\textbf{Safety:}
\begin{itemize}
  \item LLM input guard (injection detection)
  \item Output sanitization
  \item Iteration limits (max 5/8)
  \item Request timeouts (30s/60s)
\end{itemize}

\vspace{0.5em}
\textbf{Auth:}
\begin{itemize}
  \item JWT tokens (24h expiry)
  \item Role-based access (user/admin)

\column{0.5\textwidth}
\textbf{Session Management:}
\begin{itemize}
  \item MongoDB Atlas persistence
  \item 4-message context window
  \item Conversation rename/delete
\end{itemize}

\vspace{0.5em}
\textbf{Performance:}
\begin{itemize}
  \item Redis semantic cache
  \item BERT fast-path routing
